cmake_minimum_required(VERSION 3.20)
project(ailocal_ct2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CT2_SOURCE OR NOT CT2_BUILD)
  message(FATAL_ERROR "CT2_SOURCE and CT2_BUILD are required; use scripts/setup-ctranslate2.ps1")
endif()
add_library(ailocal_ct2 SHARED bridge.cpp)
target_include_directories(ailocal_ct2 PRIVATE "${CT2_SOURCE}/include")
target_link_libraries(ailocal_ct2 PRIVATE "${CT2_BUILD}/Release/ctranslate2.lib")
target_compile_options(ailocal_ct2 PRIVATE /W4 /EHsc /utf-8)
